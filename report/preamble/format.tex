% ============================================================================
% FORMAT CHAPTER VÀ TIÊU ĐỀ CHUẨN THEO YÊU CẦU (TITLESEC)
% Kiểu: "2.   CƠ SỞ LÝ THUYẾT" (cùng 1 dòng, canh giữa, in hoa đậm)
% ============================================================================
\titleformat{\chapter}[block]
  {\normalfont\LARGE\bfseries\centering}
  {\thechapter.\quad}
  {0pt}
  {\MakeUppercase}

\titleformat{name=\chapter,numberless}[block]
  {\normalfont\LARGE\bfseries\centering}
  {}
  {0pt}
  {\MakeUppercase}

\titlespacing*{\chapter}{0pt}{10pt}{20pt}
\titlespacing*{name=\chapter,numberless}{0pt}{10pt}{20pt}

% Section: "2.1.   Tên Section" (có dấu chấm sau số mục)
\titleformat{\section}
  {\normalfont\Large\bfseries}
  {\thesection.\quad}
  {0pt}
  {}
\titlespacing*{\section}{0pt}{14pt}{6pt}

% Subsection: "2.1.1.   Tên Subsection"
\titleformat{\subsection}
  {\normalfont\large\bfseries}
  {\thesubsection.\quad}
  {0pt}
  {}
\titlespacing*{\subsection}{0pt}{10pt}{4pt}

% Subsubsection: "2.1.1.1.   Tên Subsubsection"
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries}
  {\thesubsubsection.\quad}
  {0pt}
  {}
\titlespacing*{\subsubsection}{0pt}{8pt}{3pt}

% Subsubsubsection
\makeatletter
\newcounter {subsubsubsection}[subsubsection]
\renewcommand\thesubsubsubsection{\thesubsubsection .\@alph\c@subsubsubsection}

                                     {-3.25ex\@plus -1ex \@minus -.2ex}%
                                     {1.5ex \@plus .2ex}%
                                     {\normalfont\normalsize\bfseries}}
\newcommand*\l@subsubsubsection{\@dottedtocline{3}{10.0em}{4.1em}}
\makeatother
